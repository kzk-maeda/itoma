const { chromium }=require('playwright');
const fs=require('fs');const path=require('path');
(async()=>{
const out=process.env.ITOMA_SITE_QA_OUTPUT||'/workspace/itoma/docs/screenshots/service-site';fs.mkdirSync(out,{recursive:true});
const base=process.env.ITOMA_SITE_QA_BASE||'http://127.0.0.1:8800/';
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
const c=await browser.newContext({viewport:{width:1440,height:1000},reducedMotion:'reduce'});c.setDefaultTimeout(15000);const p=await c.newPage();let errors=[],external=[];p.on('pageerror',e=>errors.push(e.message));p.on('request',r=>{if(!r.url().startsWith(base))external.push(r.url());});
const assert=(ok,message)=>{if(!ok)throw Error(message);};
const pages=['service-site.html','privacy-guide.html','support-guide.html'];const cases=[];
try{
await p.goto(base+(process.env.ITOMA_SITE_QA_ROOT?'':pages[0]),{waitUntil:'networkidle'});await p.getByRole('heading',{name:/スマホを置いて、.*自分の時間へ。/}).waitFor();
await p.screenshot({path:path.join(out,'home-desktop.png'),fullPage:true});await p.screenshot({path:path.join(out,'home-first-screen.png')});
await p.getByRole('link',{name:'いとまの使い方'}).click();await p.waitForFunction(()=>location.hash==='#how-it-works');
await p.getByRole('link',{name:'プライバシーポリシーを読む'}).click();await p.getByRole('heading',{name:'プライバシーポリシー',exact:true}).waitFor();
await p.getByRole('navigation',{name:'このページの目次'}).getByRole('link',{name:'4. 記録の削除、設定変更、許可の取消',exact:true}).click();await p.waitForFunction(()=>location.hash==='#deletion');
await p.getByRole('link',{name:'記録の削除について確認する'}).click();await p.getByRole('heading',{name:/迷ったときに、.*戻れる場所。/}).waitFor();
const faq=p.getByText('理由だけを削除すると、何が残りますか？',{exact:true});await faq.click();assert(await p.locator('details[open]').count()===1,'FAQ did not open');await faq.click();
await p.getByRole('button',{name:'問い合わせ窓口の準備状況',exact:true}).click();await p.getByRole('dialog').waitFor();await p.keyboard.press('Escape');assert(!(await p.getByRole('dialog').isVisible()),'Escape failed');assert(await p.getByRole('button',{name:'問い合わせ窓口の準備状況',exact:true}).evaluate(el=>el===document.activeElement),'dialog focus not restored');
await p.getByRole('button',{name:'問い合わせ窓口の準備状況',exact:true}).click();await p.getByRole('button',{name:'閉じる',exact:true}).click();
await p.goBack();await p.getByRole('heading',{name:'プライバシーポリシー',exact:true}).waitFor();
for(const page of pages){
 await p.goto(base+page,{waitUntil:'networkidle'});
 const links=await p.locator('a[href]').evaluateAll(es=>es.map(e=>e.getAttribute('href')));
 for(const href of new Set(links)){
  const url=new URL(href,p.url());if(url.origin!==new URL(base).origin){assert(url.protocol==='https:'&&url.hostname==='www.apple.com','unapproved external link '+href);continue;}
  const res=await p.request.get(url.href.split('#')[0]);assert(res.ok(),'broken link '+url.href);
  if(url.hash){const html=await res.text();assert(html.includes('id="'+decodeURIComponent(url.hash.slice(1))+'"'),'missing anchor '+href);}
 }
 assert(await p.locator('form,input,textarea').count()===0,'unexpected user data collection');
 assert(await p.locator('h1').count()===1,'not single h1');
 if(page==='privacy-guide.html'){
  assert(await p.locator('.prose section h2').count()===9,'policy chapters missing');
  const normalize=s=>s.replace(/\s/g,'');
  const rendered=normalize(await p.locator('.prose').innerText());
  const md=fs.readFileSync('/workspace/itoma/docs/privacy-policy-draft.md','utf8');
  for(const line of md.split('\n')){
   if(!line.trim()||line.startsWith('#')||line.startsWith('|'))continue;
   const plain=line.replace(/^[-] /,'').replace(/\[([^\]]+)\]\(https:\/\/[^)]+\)/g,'$1').replace(/\*\*/g,'').trim();
   assert(rendered.includes(normalize(plain)),'missing source paragraph: '+plain.slice(0,65));
  }
  for(const row of md.split('\n').filter(l=>l.startsWith('|')&&!l.includes('---'))){for(const cell of row.split('|').slice(1,-1)){assert(rendered.includes(normalize(cell)),'missing source table cell: '+cell.slice(0,60));}}
 }

 await p.screenshot({path:path.join(out,page.replace('.html','')+'-desktop.png'),fullPage:true});
 for(const width of [320,390,768,1440]){
  await p.setViewportSize({width,height:1000});
  for(const size of [16,32]){
   await p.evaluate(size=>document.documentElement.style.fontSize=size+'px',size);
   const m=await p.evaluate(()=>({width:innerWidth,scroll:document.documentElement.scrollWidth}));assert(m.scroll<=m.width,'overflow '+page+' '+width+' '+size+' '+JSON.stringify(m));
   const min=await p.locator('header a,footer a,.button,summary,.toc a,.plain-link').evaluateAll(es=>Math.min(...es.filter(e=>e.getBoundingClientRect().height>0).map(e=>e.getBoundingClientRect().height)));assert(min>=44,'tap area '+page+' '+min);
   if(width===390&&size===16)await p.screenshot({path:path.join(out,page.replace('.html','')+'-mobile.png'),fullPage:true});
   if(width===320&&size===32)await p.screenshot({path:path.join(out,page.replace('.html','')+'-text200.png'),fullPage:true});
   cases.push({page,width,textPercent:size/16*100});
  }
 }
 await p.evaluate(()=>document.documentElement.style.fontSize='16px');await p.setViewportSize({width:1440,height:1000});await p.emulateMedia({colorScheme:'dark',reducedMotion:'reduce'});await p.screenshot({path:path.join(out,page.replace('.html','')+'-dark.png'),fullPage:true});await p.emulateMedia({colorScheme:'light',reducedMotion:'reduce'});
}
await p.goto(base+'service-site.html');await p.keyboard.press('Tab');assert(await p.getByRole('link',{name:'本文へ移動',exact:true}).evaluate(el=>el===document.activeElement),'skip link focus failed');await p.keyboard.press('Enter');assert(new URL(p.url()).hash==='#main','skip link failed');
await p.goto(base+'support-guide.html');const first=p.locator('summary').first();await first.focus();await p.keyboard.press('Enter');assert(await p.locator('details[open]').count()===1,'keyboard FAQ failed');
assert(!errors.length,JSON.stringify(errors));assert(!external.length,JSON.stringify(external));
fs.writeFileSync(path.join(out,'verification.json'),JSON.stringify({kind:'service-site-static-study',baseMain:'d57c68e',nativeOrOSVerified:false,browser:await browser.version(),cases,errors,externalRequests:external},null,2)+'\n');
console.log('PASS: 3 pages, all local links/anchors, top-root, privacy/support flow, browser back, FAQ mouse/keyboard, dialog close/Escape/focus, skip link, 320/390/768/1440, text200%, dark/reduce-motion, 44px, no external requests/forms/page errors');
}finally{await browser.close();}
})();
