const { chromium }=require('playwright');
const fs=require('fs'),path=require('path');
(async()=>{
const out=process.env.ITOMA_SITE_QA_OUTPUT||'/workspace/itoma/docs/screenshots/service-site';fs.mkdirSync(out,{recursive:true});
const base=process.env.ITOMA_SITE_QA_BASE||'http://127.0.0.1:8800/';
const browser=await chromium.launch({executablePath:'/usr/bin/chromium',args:['--no-sandbox']});
const c=await browser.newContext({viewport:{width:1440,height:1000},reducedMotion:'reduce'});c.setDefaultTimeout(15000);
const p=await c.newPage();let errors=[],external=[];p.on('pageerror',e=>errors.push(e.message));p.on('request',r=>{if(!r.url().startsWith(base))external.push(r.url());});
const assert=(ok,msg)=>{if(!ok)throw Error(msg);};const pages=['service-site.html','privacy-guide.html'],cases=[];
try{
await p.goto(base+(process.env.ITOMA_SITE_QA_ROOT?'':pages[0]),{waitUntil:'networkidle'});
await p.getByRole('heading',{name:/スマホを置いて、.*自分の時間へ。/}).waitFor();
assert(await p.locator('main section').count()===2,'LP must only have hero and usage');
assert(await p.locator('main .step').count()===3,'missing usage steps');
assert(await p.locator('details,dialog,form,input,textarea,script').count()===0,'unneeded interactions or runtime');
assert(await p.locator('header nav').count()===0,'unneeded header navigation');
await p.screenshot({path:path.join(out,'home-desktop.png'),fullPage:true});await p.screenshot({path:path.join(out,'home-first-screen.png')});
await p.getByRole('link',{name:'いとまの使い方'}).click();await p.waitForFunction(()=>location.hash==='#how-it-works');
await p.locator('footer').getByRole('link',{name:'プライバシーポリシー',exact:true}).click();await p.getByRole('heading',{name:'プライバシーポリシー',exact:true}).waitFor();
await p.getByRole('navigation',{name:'このページの目次'}).getByRole('link',{name:'4. 記録の削除、設定変更、許可の取消',exact:true}).click();await p.waitForFunction(()=>location.hash==='#deletion');
await p.goBack();await p.goBack();await p.getByRole('heading',{name:/スマホを置いて、.*自分の時間へ。/}).waitFor();
for(const page of pages){
 await p.goto(base+page,{waitUntil:'networkidle'});
 assert(await p.locator('a[target="_blank"]').count()===0,'unexpected new tab');
 assert(await p.locator('footer a[href="mailto:support@itoma.app"]').count()===1,'missing public support email');
 const links=await p.locator('a[href]').evaluateAll(es=>es.map(e=>e.getAttribute('href')));
 for(const href of new Set(links)){
  if(href.startsWith('mailto:')){assert(href==='mailto:support@itoma.app','wrong mailbox');continue;}
  const url=new URL(href,p.url());
  if(url.origin!==new URL(base).origin){assert(url.protocol==='https:'&&url.hostname==='www.apple.com','unapproved external link');continue;}
  assert(!url.pathname.includes('support-guide'),'support page still linked');
  const res=await p.request.get(url.href.split('#')[0]);assert(res.ok(),'broken link '+href);
  if(url.hash)assert((await res.text()).includes('id="'+decodeURIComponent(url.hash.slice(1))+'"'),'missing anchor '+href);
 }
 assert(await p.locator('h1').count()===1,'not single h1');
 assert(await p.locator('form,input,textarea,dialog,script').count()===0,'unexpected interactive collection');
 if(page==='privacy-guide.html'){
  assert(await p.locator('.prose section h2').count()===9,'policy chapters missing');
  const normalize=s=>s.replace(/\s/g,'');const rendered=normalize(await p.locator('.prose').innerText());
  const md=fs.readFileSync('/workspace/itoma/docs/privacy-policy-draft.md','utf8');
  for(const line of md.split('\n')){if(!line.trim()||line.startsWith('#')||line.startsWith('|'))continue;const plain=line.replace(/^[-] /,'').replace(/\[([^\]]+)\]\(https:\/\/[^)]+\)/g,'$1').replace(/\*\*/g,'').trim();assert(rendered.includes(normalize(plain)),'missing source paragraph '+plain.slice(0,65));}
  for(const row of md.split('\n').filter(l=>l.startsWith('|')&&!l.includes('---')))for(const cell of row.split('|').slice(1,-1))assert(rendered.includes(normalize(cell)),'missing source table cell');
 }
 await p.screenshot({path:path.join(out,page.replace('.html','')+'-desktop.png'),fullPage:true});
 for(const width of [320,390,768,1440]){
  await p.setViewportSize({width,height:1000});
  for(const size of [16,32]){
   await p.evaluate(size=>document.documentElement.style.fontSize=size+'px',size);
   assert(await p.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),'overflow '+page+' '+width+'/'+size);
   const min=await p.locator('header a,footer a,.button,.toc a,.plain-link').evaluateAll(es=>Math.min(...es.filter(e=>e.getBoundingClientRect().height>0).map(e=>e.getBoundingClientRect().height)));assert(min>=44,'tap height '+min);
   if(width===390&&size===16)await p.screenshot({path:path.join(out,page.replace('.html','')+'-mobile.png'),fullPage:true});
   if(width===320&&size===32)await p.screenshot({path:path.join(out,page.replace('.html','')+'-text200.png'),fullPage:true});
   cases.push({page,width,textPercent:size/16*100});
  }
 }
 await p.evaluate(()=>document.documentElement.style.fontSize='16px');await p.setViewportSize({width:1440,height:1000});await p.emulateMedia({colorScheme:'dark',reducedMotion:'reduce'});await p.screenshot({path:path.join(out,page.replace('.html','')+'-dark.png'),fullPage:true});await p.emulateMedia({colorScheme:'light',reducedMotion:'reduce'});
}
await p.goto(base+'service-site.html');await p.keyboard.press('Tab');assert(await p.getByRole('link',{name:'本文へ移動',exact:true}).evaluate(el=>el===document.activeElement),'skip focus failed');await p.keyboard.press('Enter');assert(new URL(p.url()).hash==='#main','skip link failed');
assert(!errors.length,JSON.stringify(errors));assert(!external.length,JSON.stringify(external));
fs.writeFileSync(path.join(out,'verification.json'),JSON.stringify({kind:'minimal-LP-and-policy',physicalDevice:false,browser:await browser.version(),cases,errors,externalRequests:external},null,2)+'\n');
console.log('PASS: minimal hero/usage/footer, 2 pages, 9-chapter policy/source parity, links/anchors/browser back, support mailto without sending, no support page/new tab/forms/JS, 16 viewport-text cases, dark/reduce-motion/keyboard, 44px, no overflow/errors/remote requests');
}finally{await browser.close();}
})();
