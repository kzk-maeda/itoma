import { createRequire } from 'node:module';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
const require = createRequire(import.meta.url);
const { chromium } = require('playwright');
const output = '/tmp/itoma-demo-aligned-ui/docs/qa/demo-aligned-ui';
const browser = await chromium.launch({ executablePath: '/usr/bin/chromium', headless: true, args: ['--no-sandbox'] });
const cases = [
  ['home-light', '/', 'observed', 'light', 390, 844, false],
  ['home-dark', '/', 'observed', 'dark', 390, 844, false],
  ['home-se', '/', 'observed', 'light', 375, 667, false],
  ['home-320-large', '/', 'observed', 'light', 320, 800, true],
  ['home-empty', '/', 'empty', 'light', 390, 844, false],
  ['home-uncertain', '/', 'uncertain', 'dark', 390, 844, false],
  ['home-grant', '/', 'grant', 'light', 390, 844, false],
  ['reflection-light', '/reflection', 'observed', 'light', 390, 844, false],
  ['reflection-dark', '/reflection', 'observed', 'dark', 390, 844, false],
  ['reflection-se', '/reflection', 'observed', 'light', 375, 667, false],
  ['reflection-320-large', '/reflection', 'observed', 'dark', 320, 800, true],
  ['reflection-deleted', '/reflection', 'deleted', 'light', 390, 844, false],
  ['reflection-unrecorded', '/reflection', 'notRecorded', 'light', 390, 844, false],
  ['reflection-error', '/reflection', 'error', 'light', 390, 844, false],
  ['settings-light', '/settings', 'observed', 'light', 390, 844, false],
  ['settings-dark', '/settings', 'observed', 'dark', 390, 844, false],
  ['settings-320-large', '/settings', 'observed', 'light', 320, 800, true],
];
const results = { kind: 'synthetic-native-ports-web-product-routes', physicalDevice: false, osRestrictionsVerified: false,
  browser: await browser.version(), fixedClock: '2026-11-09T21:30:00Z', reducedMotion: 'reduce', cases: [] };
const makeContext = async (width, height, colorScheme) => {
  const context = await browser.newContext({ viewport: { width, height }, colorScheme, reducedMotion: 'reduce' });
  await context.addInitScript(at => { Date.now = () => at; }, Date.parse(results.fixedClock));
  return context;
};
try {
  for (const [name, route, scenario, colorScheme, width, height, large] of cases) {
    const context = await makeContext(width, height, colorScheme), page = await context.newPage(), errors = [];
    page.on('pageerror', e => errors.push(e.message.slice(0, 200)));
    page.on('console', m => { if (m.type() === 'error') errors.push(m.text().slice(0, 200)); });
    await page.goto(`http://127.0.0.1:8161${route}?scenario=${scenario}`, { waitUntil: 'networkidle' });
    await page.getByRole('heading', { name: route === '/' ? 'いとま' : route === '/settings' ? 'ルールと設定' : '時間の積み上げ', exact: true }).waitFor();
    if (route === '/') {
      if (scenario !== 'empty') await page.getByTestId('period-ring').waitFor();
    } else if (route === '/reflection') {
      await page.getByText('時間の集計は準備中です', { exact: true }).waitFor();
      if (scenario !== 'error') {
        await page.getByRole('button', { name: '月次', exact: true }).click();
        await page.getByRole('heading', { name: '2026年11月', exact: true }).waitFor();
      }
      if (await page.getByTestId('reflection-total').innerText() !== '—') throw new Error('Product invented a duration');
    }
    if (large) await page.evaluate(() => {
      const nodes = [...document.querySelectorAll('[dir="auto"]')].map(el => [el, getComputedStyle(el)]).map(([el, s]) => [el, parseFloat(s.fontSize), parseFloat(s.lineHeight)]);
      for (const [el, font, line] of nodes) { el.style.fontSize = `${font * 1.5}px`; if (Number.isFinite(line)) el.style.lineHeight = `${line * 1.5}px`; }
    });
    const measurement = await page.evaluate(() => ({
      horizontalOverflow: document.documentElement.scrollWidth > innerWidth,
      minActionHeight: Math.min(...[...document.querySelectorAll('[role="button"],[role="tab"]')].map(el => el.getBoundingClientRect().height)),
      tabCount: document.querySelectorAll('[role="tab"]').length,
    }));
    await page.screenshot({ path: `${output}/${name}.png` });
    const lastAction = page.getByRole('button', { name: route === '/' ? scenario === 'empty' ? 'ルールをつくる' : '一時解除する' : route === '/settings' ? '保存した理由の削除を確認' : '全記録と累計を削除する', exact: true });
    await lastAction.scrollIntoViewIfNeeded();
    const box = await lastAction.boundingBox();
    const tabBox = await page.getByRole('tab').first().boundingBox();
    const reachable = !!box && !!tabBox && box.y >= 0 && box.y + box.height <= tabBox.y - 8;
    const entry = { name, route, scenario, colorScheme, width, height, artificialTextScale: large ? 1.5 : 1, ...measurement,
      lastActionReachableAboveTabs: reachable, errors,
      sha256: createHash('sha256').update(readFileSync(`${output}/${name}.png`)).digest('hex') };
    results.cases.push(entry);
    if (measurement.horizontalOverflow || measurement.minActionHeight < 44 || measurement.tabCount !== 3 || !reachable || errors.length) throw new Error(JSON.stringify(entry));
    await context.close();
  }
  const context = await makeContext(390, 844, 'light'), page = await context.newPage();
  await page.goto('http://127.0.0.1:8161/', { waitUntil: 'networkidle' });
  await page.getByRole('tab', { name: '振り返り', exact: true }).click();
  await page.getByRole('button', { name: '月次', exact: true }).click();
  await page.getByRole('heading', { name: '2026年11月', exact: true }).waitFor();
  await page.getByRole('button', { name: '次の期間', exact: true }).click();
  await page.getByRole('heading', { name: '2026年12月', exact: true }).waitFor();
  const initialHistoryLength = await page.evaluate(() => history.length);
  for (let n = 0; n < 100; n += 1) {
    await page.getByRole('tab', { name: 'ホーム', exact: true }).click();
    await page.getByRole('heading', { name: 'いとま', exact: true }).waitFor();
    await page.getByRole('tab', { name: '振り返り', exact: true }).click();
    await page.getByRole('heading', { name: '2026年12月', exact: true }).waitFor();
    if (await page.getByRole('tab').count() !== 3) throw new Error('Duplicate visible navigation');
  }
  const finalHistoryLength = await page.evaluate(() => history.length);
  if (finalHistoryLength > initialHistoryLength + 1) throw new Error(`History grew: ${initialHistoryLength} -> ${finalHistoryLength}`);
  results.navigation = { roundTrips: 100, rememberedMonth: '2026-12', initialHistoryLength, finalHistoryLength, tabs: 3 };
  await page.getByRole('button', { name: /^12月1日、/ }).click();
  await page.getByRole('heading', { name: '11月30日からの週', exact: true }).waitFor();
  await page.getByRole('button', { name: /^12月2日、/ }).click();
  await page.getByRole('heading', { name: 'あなたの選択の記録', exact: true }).waitFor();
  await page.getByRole('button', { name: '‹ 戻る', exact: true }).click();
  await page.getByRole('heading', { name: '11月30日からの週', exact: true }).waitFor();
  results.drilldown = { monthToClippedWeekToDayHistoryAndBack: true, privateReasonNotExercised: true };
  await context.close();
} finally {
  writeFileSync(`${output}/manifest.json`, JSON.stringify(results, null, 2) + '\n');
  await browser.close();
}
