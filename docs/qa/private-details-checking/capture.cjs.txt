async function scaleText(page, scale) {
 if(scale===1)return;
 await page.evaluate(scale=>{
  for(const element of document.querySelectorAll('[dir="auto"],input,textarea')) {
   const style=getComputedStyle(element);
   element.dataset.proofFontSize ??= style.fontSize;
   element.dataset.proofLineHeight ??= style.lineHeight;
   element.style.fontSize=`${parseFloat(element.dataset.proofFontSize)*scale}px`;
   const line=parseFloat(element.dataset.proofLineHeight);if(Number.isFinite(line))element.style.lineHeight=`${line*scale}px`;
  }
 },scale);
}

async function capture(page, name) {
  const file = `${output}/${name}.png`;
  await page.screenshot({path:file,fullPage:true});
  return {file,sha256:digest(file)};
}
