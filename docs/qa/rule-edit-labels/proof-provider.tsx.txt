export function RuleLabelProof({ children }: { children: ReactNode }) {
  const { privateState } = useLocalSearchParams<{privateState?: string}>();
  const client = useMemo(() => ({ run: async (): Promise<ScreenTimeResponse> => {
    const now = Date.now() / 1000;
    const snapshot: ScreenTimeSnapshot = { ...unavailableSnapshot('device'), storage: 'ready', sequence: '1', revision: '1',
      activation: 'enabled', authorization: 'approved', schedulesReady: true, observation: 'applied', evaluatedAt: now, validUntil: now + 1800,
      ruleDetailsStatus: privateState === 'unavailable' ? 'unavailable' : 'ready', ruleDetailsRevision: '1',

      selections: [{ ref: ruleID, count: 2 }], targets: [], periods: [{ ruleID, startsAt: now - 1800, endsAt: now + 5400 }] };
    return { status: 'ok', impacts: [], snapshot };
  } }), [privateState]);
  const onboarding = useMemo(() => ({ run: async () => ({ status: 'ok' as const, impacts: [], onboarding: { revision: '1', seenIntro: true, completed: true, selectionReady: false, canEdit: true } }) }), []);
